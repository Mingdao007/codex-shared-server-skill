cmake_minimum_required(VERSION 3.16)
project(codex_shared_server LANGUAGES CXX)
find_package(Boost REQUIRED COMPONENTS system)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(JSONCPP REQUIRED IMPORTED_TARGET jsoncpp)
add_executable(shared-app-launcher shared_app_launcher.cpp)
target_compile_features(shared-app-launcher PRIVATE cxx_std_17)
target_compile_options(shared-app-launcher PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(shared-app-launcher PRIVATE Boost::system PkgConfig::JSONCPP Threads::Threads)
